Departments var sheetDept = getOrInsertSheet(ss, "Departments"); var dataDept = sheetDept.getDataRange().getValues(); var departments = []; for (var j = 1; j < dataDept.length; j++) { if(dataDept[j][0]) { departments.push({ name: dataDept[j][0], code: dataDept[j][1] }); } } return responseJSON({ assets: assets, departments: departments }); } function saveAsset(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = getOrInsertSheet(ss, "Assets"); var rootFolder = getOrCreateFolder(DriveApp.getRootFolder(), "ประวัติอุปกรณ์สำหรับงานซ่อมบำรุง"); sheet.getRange(row, 3).setValue(data.model); sheet.getRange(row, 4).setValue(data.location); sheet.getRange(row, 5).setValue(data.status); sheet.getRange(row, 6).setValue(data.purchaseDate); if(imgUrl.startsWith("http")) sheet.getRange(row, 7).setValue(imgUrl); if(pdfUrl.startsWith("http")) sheet.getRange(row, 9).setValue(pdfUrl); sheet.getRange(row, 10).setValue(data.vendor || ""); // Col 11 (nextPmDate) ไม่ต้องแก้ที่นี่ ให้ระบบ Auto Sync จาก PM Schedule แทน } else { sheet.appendRow([ data.id, data.name, data.model, data.location, data.status, data.purchaseDate, imgUrl, "[]", pdfUrl, data.vendor||"", "" ]); } return responseJSON({ status: 'success' }); } function saveLog(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = getOrInsertSheet(ss, "Assets"); var found = sheet.createTextFinder(data.assetId).findNext(); currentHistory.unshift(data.log); historyCell.setValue(JSON.stringify(currentHistory)); return responseJSON({ status: 'success' }); } function saveDepartment(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (sheet) { var textFinder = sheet.getRange("H:H").createTextFinder(String(data.schId)); var found = textFinder.findNext(); if (found) { sheet.deleteRow(found.getRow()); SpreadsheetApp.flush(); // 🟢 NEW: อัปเดตวันที่ PM ครั้งถัดไปที่ Sheet หลักทันที (กรณีลบงานทิ้ง วันที่ต้องเปลี่ยน) updateAssetMasterPmDate(data.assetId, sheet); } } return responseJSON({ status: 'success' }); } // 🟢 NEW: ฟังก์ชันช่วยอัปเดตวันที่ PM ที่หน้าแรก function updateAssetMasterPmDate(assetId, pmSheet) { try { var data = pmSheet.getDataRange().getValues(); var today = new Date(); today.setHours(0,0,0,0); } } catch(e) { Logger.log("Error updating master date: " + e.toString()); } } // --- Helper Functions --- function getOrInsertSheet(ss, name) { var s = ss.getSheetByName(name); if (!s) { s = ss.insertSheet(name); } catch(e) { return d; } } function processFileUpload(base64Data, folder, fileNamePrefix) { if(!base64Data || !base64Data.startsWith("data")) return base64Data || ""; try{ var split=base64Data.split(',');
